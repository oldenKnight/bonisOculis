# bonis oculis analyzer - runtime dependencies (Python 3.11+). Nothing else is imported at runtime.
# Every package is pinned to an exact version (review A9). Unless noted, the version is the one
# installed and tested on the Linux dev container (2026-10-13). The list is the whole dependency
# closure, platform markers included, so the resolver has nothing left to choose:
#   python -m pip install -r analyzer/requirements.txt
# Next step (not done): hash-lock with `pip-compile --generate-hashes` per platform and install
# with --require-hashes.

# numpy 2.5 needs Python >= 3.12; CI and the build runners use 3.11 -> 2.4.6 (newest with
# cp311 wheels; NOT tested on the dev container, which runs 3.13 + 2.5.3).
numpy==2.5.3 ; python_version >= "3.12"
numpy==2.4.6 ; python_version < "3.12"

# ONNX Runtime: pinned to releases WITHOUT Microsoft's 1DS telemetry client (review A1).
# 1.29.0 is the first PyPI release that carries it on Linux and macOS (it uploads to
# mobile.events.data.microsoft.com and writes ~/.cache/Microsoft/DeveloperTools); 1.28.0 and
# older have no such code (checked in the wheels' native libraries). bonis_analyzer/__init__.py
# also sets ORT_DISABLE_TELEMETRY=1, which silences 1.29+ should one ever be installed.
#   Linux x64, macOS arm64: 1.28.0 (newest without the client)
#   macOS x64: 1.23.2 (the last release with an Intel macOS wheel; no client)
#   Windows: onnxruntime-directml 1.24.4 (newest DirectML release; no 1DS client; its ETW
#            provider is switched off with onnxruntime.disable_telemetry_events())
onnxruntime==1.28.0 ; sys_platform == "linux" or (sys_platform == "darwin" and platform_machine != "x86_64")
onnxruntime==1.23.2 ; sys_platform == "darwin" and platform_machine == "x86_64"
onnxruntime-directml==1.24.4 ; sys_platform == "win32"
flatbuffers==25.12.19
packaging==26.3
